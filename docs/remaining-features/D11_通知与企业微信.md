-- 1. Outbox 表
CREATE TABLE outbox_messages (
  id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id    UUID NOT NULL,
  event_type   TEXT NOT NULL,
  payload      JSONB NOT NULL,
  status       TEXT NOT NULL DEFAULT 'PENDING'
               CHECK (status IN ('PENDING','PUBLISHED','FAILED')),
  retry_count  INT NOT NULL DEFAULT 0,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  published_at TIMESTAMPTZ
);
CREATE INDEX ON outbox_messages(status, created_at) WHERE status = 'PENDING';

-- 2. 通知派发记录
CREATE TABLE notification_deliveries (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id       UUID NOT NULL,
  recipient_id    UUID NOT NULL REFERENCES accounts(id),
  channel         TEXT NOT NULL CHECK (channel IN ('WECOM','IN_APP','EMAIL')),
  event_type      TEXT NOT NULL,
  payload         JSONB NOT NULL,
  status          TEXT NOT NULL DEFAULT 'SENT'
                  CHECK (status IN ('SENT','FAILED')),
  external_msg_id TEXT,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX ON notification_deliveries(recipient_id, created_at);

-- 3. 企微租户配置
CREATE TABLE wecom_tenant_configs (
  tenant_id  UUID PRIMARY KEY REFERENCES tenants(id),
  corp_id    TEXT NOT NULL,
  agent_id   TEXT NOT NULL,
  secret     TEXT NOT NULL,  -- 生产建议加密
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 4. 员工企微 ID 绑定
ALTER TABLE accounts ADD COLUMN IF NOT EXISTS wecom_user_id TEXT;

### 3.1 OutboxService

**文件**：`src/main/java/com/mop/notification/OutboxService.java`
```java
@Service
@RequiredArgsConstructor
public class OutboxService {

    private final OutboxRepository repo;

    /**
     * 在业务事务内调用，与业务操作原子提交。
     * propagation = MANDATORY 确保调用方已开事务。
     */
    @Transactional(propagation = Propagation.MANDATORY)
    public void publish(String eventType, UUID referenceId, UUID tenantId) {
        repo.save(OutboxMessage.builder()
            .tenantId(tenantId)
            .eventType(eventType)
            .payload(Map.of(
                "referenceId", referenceId.toString(),
                "tenantId",    tenantId.toString()
            ))
            .status(OutboxStatus.PENDING)
            .build());
    }
}
```

### 3.3 OutboxRelayWorker

**文件**：`src/main/java/com/mop/notification/OutboxRelayWorker.java`
```java
@Component @Slf4j
public class OutboxRelayWorker {

    @Scheduled(fixedDelay = 2_000)
    @Transactional
    public void relay() {
        List<OutboxMessage> batch = repo
            .findTop100ByStatusOrderByCreatedAt(OutboxStatus.PENDING);

        for (OutboxMessage m : batch) {
            try {
                rabbit.convertAndSend(
                    "mop.notifications",
                    m.getEventType(),
                    m.getPayload()
                );
                m.setStatus(OutboxStatus.PUBLISHED);
                m.setPublishedAt(Instant.now());
            } catch (Exception e) {
                m.setRetryCount(m.getRetryCount() + 1);
                if (m.getRetryCount() >= 5)
                    m.setStatus(OutboxStatus.FAILED);
                log.error("outbox relay failed id={}", m.getId(), e);
            }
            repo.save(m);
        }
    }
}
```

### 3.4 WeComClient

**文件**：`src/main/java/com/mop/notification/WeComClient.java`
```java
@Component
public class WeComClient {

    private final RestTemplate http;

    // Token 缓存（有效期 7200s，110分钟主动就新）
    private final Cache<String, String> tokenCache =
        Caffeine.newBuilder().expireAfterWrite(110, MINUTES).build();

    public String getToken(String corpId, String secret) {
        return tokenCache.get(corpId, id -> {
            var resp = http.getForObject(
                "https://qyapi.weixin.qq.com/cgi-bin/gettoken?corpid={c}&corpsecret={s}",
                Map.class, corpId, secret);
            if (resp == null || !"ok".equals(resp.get("errmsg")))
                throw new WeComException("获取Token失败: " + resp);
            return (String) resp.get("access_token");
        });
    }

    /** 发送图文卡片消息，返回 msgid */
    public String sendTextCard(String token, String agentId,
                               String toUser, String title,
                               String desc, String url) {
        var body = Map.of(
            "touser",  toUser,
            "msgtype", "textcard",
            "agentid", Integer.parseInt(agentId),
            "textcard", Map.of(
                "title",       title,
                "description", desc,
                "url",         url
            )
        );
        var resp = http.postForObject(
            "https://qyapi.weixin.qq.com/cgi-bin/message/send?access_token=" + token,
            body, Map.class);
        if (resp == null || !Integer.valueOf(0).equals(resp.get("errcode")))
            throw new WeComException("发送失败: " + resp);
        return (String) resp.get("msgid");
    }
}
```

### 3.5 WeComSenderWorker

**文件**：`src/main/java/com/mop/notification/WeComSenderWorker.java`
```java
@Component @Slf4j
public class WeComSenderWorker {

    @RabbitListener(queues = "mop.wecom")
    public void handle(
            @Payload Map<String, Object> payload,
            @Header(AmqpHeaders.RECEIVED_ROUTING_KEY) String routingKey) {

        UUID refId    = UUID.fromString((String) payload.get("referenceId"));
        UUID tenantId = UUID.fromString((String) payload.get("tenantId"));

        WecomConfig cfg = configRepo.findByTenantId(tenantId).orElse(null);
        if (cfg == null) {
            log.warn("租户无企微配置: {}", tenantId); return;
        }

        String token = weComClient.getToken(cfg.getCorpId(), cfg.getSecret());
        List<Account> recipients = resolveRecipients(routingKey, refId, tenantId);

        for (Account r : recipients) {
            if (r.getWecomUserId() == null) continue;
            try {
                var msg  = buildMessage(routingKey, refId);
                var msgId = weComClient.sendTextCard(
                    token, cfg.getAgentId(), r.getWecomUserId(),
                    msg.title(), msg.desc(), msg.url());
                save(tenantId, r.getId(), routingKey, payload, "SENT", msgId);
            } catch (Exception e) {
                log.error("企微发送失败 recipient={}", r.getId(), e);
                save(tenantId, r.getId(), routingKey, payload, "FAILED", null);
            }
        }
    }

    private record Msg(String title, String desc, String url) {}

    private Msg buildMessage(String eventType, UUID refId) {
        return switch (eventType) {
            // 工单相关
            case "ticket.accepted" -> {
                String link = appBaseUrl + "/tickets/" + refId;
                yield new Msg("📥 工单已接单", "您的报障已被运维员接单，请监控进展", link);
            }
            case "ticket.resolved" -> {
                String link = appBaseUrl + "/tickets/" + refId;
                yield new Msg("✅ 工单已解决", "运维员已标记解决，请点击确认并关闭工单", link);
            }
            case "ticket.sla_breached" -> {
                String link = appBaseUrl + "/tickets/" + refId;
                yield new Msg("⚠️ SLA 超期", "工单已超过 SLA 时限，请尽快处理", link);
            }
            case "ticket.reopened" -> {
                String link = appBaseUrl + "/tickets/" + refId;
                yield new Msg("🔄 工单被重开", "客户拒绝了解决方案，工单重开处理中", link);
            }
            // 变更审批相关
            case "change.approved" -> {
                String link = appBaseUrl + "/change-notices/" + refId;
                yield new Msg("✅ 变更已批准", "您的变更报备已审批通过，可在窗口期内执行", link);
            }
            case "change.rejected" -> {
                String link = appBaseUrl + "/change-notices/" + refId;
                yield new Msg("❌ 变更被拒绝", "您的变更报备已被拒绝，请查看拒绝原因", link);
            }
            // 巡检相关
            case "inspection.completed" -> {
                String link = appBaseUrl + "/inspection-worksheets/" + refId;
                yield new Msg("📝 巡检已完成", "巡检工单已完成，请查看结果报告", link);
            }
            // 月报/季报相关
            case "report.ready", "report.quarterly_ready" -> {
                GeneratedReport report = reportRepo.findLatestByTenantId(resolveTenanId(refId));
                String link = report != null ? report.getDownloadUrl() :
                    appBaseUrl + "/reports";
                String type = eventType.contains("quarterly") ? "季报" : "月报";
                yield new Msg("📄 " + type + "已生成",
                    type + "已就绪，点击下载", link);
            }
            // 导入完成
            case "import.completed" -> {
                String link = appBaseUrl + "/customers/import";
                yield new Msg("📥 导入已完成", "批量导入作业已完成，点击查看结果", link);
            }
            // 容量告警
            case "alert.capacity" -> {
                String link = appBaseUrl + "/monitoring/capacity";
                yield new Msg("📈 容量告警", "检测到设备容量超阈值，请尽快处理", link);
            }
            // 圆屏机审批
            case "bastion.session.approved" -> {
                String link = appBaseUrl + "/bastion/terminal/" + refId;
                yield new Msg("🔐 圆屏机已审批", "连接申请已审批通过，点击连接", link);
            }
            default -> {
                String link = appBaseUrl;
                yield new Msg(eventType, refId.toString(), link);
            }
        };
    }

    /** 根据事件类型决定通知接收人 */
    private List<Account> resolveRecipients(String eventType, UUID refId, UUID tenantId) {
        return switch (eventType) {
            // 工单已解决 — 通知报障客户
            case "ticket.resolved","ticket.sla_breached" ->
                accountRepo.findTicketReporter(refId, tenantId);
            // 工单已接单 — 通知被指派运维员
            case "ticket.accepted" ->
                accountRepo.findTicketAssignee(refId, tenantId);
            default -> List.of();
        };
    }
}
```

### 合同要求（M4-13、M8-1～M8-3、M8-10、M8-11）

- **渠道配置**：端点、密钥（只读、仅轮换）、签名方式、重试预算与退避（Webhook 示例 5 次：30 秒 / 2 分钟 / 10 分钟 / 30 分钟 / 2 小时）、目标范围、域名白名单；拒绝私网地址与重定向；回调签名防循环。
- **渠道×级别矩阵**（配置项）：灾难 = 企微或钉钉 + 邮件 + 短信 + 电话 + 自动建单；严重 = 企微或钉钉 + 邮件 + 短信；一般 = 企微或钉钉 + 邮件（聚合 5 分钟）；警告 = 仅企微或钉钉；信息 = 只记录。严重级到 P 级的映射待 Q14。
- **去重键** =（实例、事件 ID、渠道、接收对象、事件阶段）；同一去重键 5 分钟内只发一次。
- **送达记录 8 列**：时间 / 事件 / 渠道 / 接收对象 / 状态 / 尝试次数 / 耗时 / 请求与回执；受理到送达 P95 ≤ 30 秒；送达率只统计 delivered。
- **死信重放**：同一去重键、新请求号；短信按月限额（`notification.sms_month_quota`，Q18）。
- **分级推送**（M8-11）：P1 = 电话外呼 + 企微（30 秒内发起）；P2 = 企微 + 短信；P3 / P4 = 企微。语音通道需甲方采购（Q18）。
- **偏好与静音**（M8-10）：静音时段内 P3 不推送但完整记录，P1 照常推送。
